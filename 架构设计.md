# 1. 项目简介
高性能数据库(前端),基于（React + Vite + Tailwind + TypeScript）
## 1.1 核心目标
安全地管理多种数据库连接（MySQL/PostgreSQL/SQLite/ClickHouse等），并支持执行 SQL 查询

## 1.2 TypeScript 类型（types/index.ts）
```TypeScript
export type DbType = 'mysql' | 'postgres' | 'sqlite';

export interface ConnectionConfig {
  id: string;
  name: string;
  type: DbType;
  host?: string;
  port?: number;
  username?: string;
  password?: string;
  database?: string;
  filePath?: string; // SQLite
}

export interface ConnectionStatus {
  id: string;
  isConnected: boolean;
  errorMessage?: string;
}
```
前后端类型要严格对齐！建议把 DbType 枚举值写成字符串字面量。

## 1.3 自定义 Hook 示例（hooks/useConnections.ts）
```TypeScript
import { useState, useEffect } from 'react';
import { fetchConnections, testConnection } from '../services/connectionApi';
import { ConnectionConfig, ConnectionStatus } from '../types';

export const useConnections = () => {
  const [connections, setConnections] = useState<ConnectionConfig[]>([]);
  const [statuses, setStatuses] = useState<Record<string, ConnectionStatus>>({});

  useEffect(() => {
    loadConnections();
  }, []);

  const loadConnections = async () => {
    const data = await fetchConnections();
    setConnections(data);
    
    // 并行测试所有连接状态
    const statusMap: Record<string, ConnectionStatus> = {};
    await Promise.all(
      data.map(async (conn) => {
        const status = await testConnection(conn.id);
        statusMap[conn.id] = status;
      })
    );
    setStatuses(statusMap);
  };

  return { connections, statuses, reload: loadConnections };
};
```
